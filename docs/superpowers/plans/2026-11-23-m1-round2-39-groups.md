#!/usr/bin/env python3
"""M1 39-group prepared-input glue. Reuses 005 prepare_view; no new segmentation."""
import argparse, json, sys
from pathlib import Path
import cv2, numpy as np, pandas as pd

WS = Path('/home/liu/ring_first_3d_workspace')
RAW = WS/'smart_endoscope_20260805 (1)'/'raw'
REG = Path('/home/liu/new-polyp-dxy/validation/frozen39/inputs.csv')
T005 = WS/'experiments/TASK-E4-UNIFIED-RING-CONTOUR-REFINEMENT-005'
INIT = Path('/home/liu/polyp_research/experiments/dxy_remeasurement_20260917/02_segmentation/model/model.safetensors')
INIT_SHA = '981aa596e8906edc4457219581c606d4a5a22bfc6f7e4924d3d9a4b9aff64464'
sys.path.insert(0, str(T005/'code'))
from prepare import prepare_view, sha, K


def vid_of(name):
    return Path(name).stem.split('_')[-1]


def reference_photos(group):
    return sorted(p.name for p in RAW.glob(f'{group}_*.jpg'))


def build_registry():
    d = pd.read_csv(REG)
    required = {'group', 'video', 'frame_index', 'stem', 'ring_mm'}
    missing = required - set(d.columns)
    if missing:
        raise ValueError(f'frozen39 missing columns: {sorted(missing)}')
    groups = sorted(d['group'].unique().tolist())
    if len(groups) != 39:
        raise ValueError(f'expected 39 groups, found {len(groups)}')
    rows, photos = [], {}
    for (g, v), sub in d.groupby(['group', 'video'], sort=True):
        raw = RAW/v
        if not raw.exists():
            raise FileNotFoundError(raw)
        rows.append(dict(group=g, video=v, frames=len(sub), raw_path=str(raw),
                         ring_mm=float(sub['ring_mm'].iloc[0]),
                         max_frame=int(sub['frame_index'].max())))
        photos[g] = reference_photos(g)
    return groups, rows, photos


def prepare_one(group, video, frames, out_root):
    video_path = RAW/video
    out = Path(out_root)/f'{group}_{vid_of(video)}'
    out.mkdir(parents=True, exist_ok=True)
    views = []
    for _, r in frames.sort_values('frame_index').iterrows():
        cache = out/f'{r.stem}.npz'
        prepare_view(video_path, int(r.frame_index), cache)
        views.append(dict(view=str(r.stem), raw_frame=int(r.frame_index), cache=str(cache)))
    meta = dict(group=group, video=video, video_path=str(video_path),
                video_sha256=sha(video_path),
                init_weights=str(INIT), init_weight_sha256=INIT_SHA,
                math='PIL bilinear 256 /255; bilinear logits upsample; argmax',
                ring_mm=float(frames['ring_mm'].iloc[0]), K=np.asarray(K).tolist(),
                views=views)
    (out/'PREPARED.json').write_text(json.dumps(meta, indent=1))
    return meta


def coverage_row(group, rows, photos, prepared_root):
    g_rows = [r for r in rows if r['group'] == group]
    ring = g_rows[0]['ring_mm']
    prepared = all((Path(prepared_root)/f"{r['group']}_{vid_of(r['video'])}"/'PREPARED.json').exists()
                   for r in g_rows)
    return dict(group_id=group,
                video_ids=';'.join(r['video'] for r in g_rows),
                reference_photo_ids=';'.join(photos[group]),
                ring_type=f'R{int(ring)}', ring_dimensions=f'inner={ring}mm;centerline={ring/2+0.6}mm;wire=0.6mm',
                raw_input_exists=all(Path(r['raw_path']).exists() for r in g_rows),
                prepared_input_exists=prepared,
                input_status='VALID' if prepared else 'PREPARED_MISSING')


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--output-root', required=True)
    ap.add_argument('--coverage', required=True)
    ap.add_argument('--groups', default='all')
    a = ap.parse_args()
    groups, rows, photos = build_registry()
    use = groups if a.groups == 'all' else a.groups.split(',')
    out_root = Path(a.output_root)
    for g in use:
        for r in [x for x in rows if x['group'] == g]:
            sub = pd.read_csv(REG)
            sub = sub[(sub.group == g) & (sub.video == r['video'])]
            prepare_one(g, r['video'], sub, out_root)
    cov = [coverage_row(g, rows, photos, out_root) for g in groups]
    Path(a.coverage).parent.mkdir(parents=True, exist_ok=True)
    pd.DataFrame(cov).to_csv(a.coverage, index=False)
    print(pd.DataFrame(cov).input_status.value_counts().to_dict())


if __name__ == '__main__':
    main()
```

def ring_candidates(arrays, ordered, root, ring_radius=5.6, ring_wire=.6):
    rings = []
    for v in ordered:
        cid = v['meta']['id']; a = arrays[cid]
        ob = observe(a['labels'])

        np.savez_compressed(root / 'observations' / (cid + '_ring.npz'), **ob)
        rawpath = root / 'predictions' / (cid + '_RING_ONLY.json')
        if rawpath.exists():
            raw = json.loads(rawpath.read_text())
        else:
            raw = recover(a['labels'], a['K'], rgb_geometry=None, initial_pose=None,
                          ring_radius=ring_radius, ring_wire=ring_wire)

#!/usr/bin/env python3
"""39-group M1 runner: discover -> prepare check -> W0 per video -> status -> aggregate."""
import argparse, json, subprocess, sys
from pathlib import Path
import pandas as pd

WS = Path('/home/liu/ring_first_3d_workspace')
RAW = WS/'smart_endoscope_20260805 (1)'/'raw'
REG = Path('/home/liu/new-polyp-dxy/validation/frozen39/inputs.csv')
PY = sys.executable
REQUIRED = ('camera_result.npz', 'accepted_cameras.npz', 'admission.csv',
            'quality_summary.json', 'manifest.json')


def classify(rc, stdout, stderr, outputs_ok, status_file_status):
    if status_file_status in ('INSUFFICIENT_EVIDENCE', 'BLOCKED_QUALITY_GATE'):
        return status_file_status
    if 'INSUFFICIENT_RING_OBSERVATION' in stderr or 'RING_FIRST_INCOMPLETE' in stderr:
        return 'INSUFFICIENT_EVIDENCE'
    if ('INSUFFICIENT_TRACKS' in stderr or 'HELD_TRACKS' in stderr
            or 'insufficient positive Ring-initialized scene points' in stderr):
        return 'BLOCKED_QUALITY_GATE'
    if not outputs_ok:
        return 'IMPLEMENTATION_ERROR'
    if rc == 0 and status_file_status in (None, 'COMPLETED_VALID', 'COMPLETED_LIMITED'):
        return status_file_status or 'COMPLETED_VALID'
    return 'IMPLEMENTATION_ERROR'


def run_video(group, video, prepared_root, out_root, work_root):
    out = Path(out_root)/group/video
    work = Path(work_root)/group/video
    out.mkdir(parents=True, exist_ok=True); work.mkdir(parents=True, exist_ok=True)
    cmd = [PY, str(WS/'tools/run_w0_camera.py'), '--group', group, '--video', video,
           '--stage', 'all', '--prepared-root', str(prepared_root),
           '--output', str(out), '--workdir', str(work)]
    r = subprocess.run(cmd, cwd=WS, capture_output=True, text=True)
    (out/'stdout.log').write_text(r.stdout); (out/'stderr.log').write_text(r.stderr)
    outputs_ok = all((out/f).exists() for f in REQUIRED)
    status_file = out/'STATUS.json'
    sf = json.loads(status_file.read_text()) if status_file.exists() else {}
    status = classify(r.returncode, r.stdout, r.stderr, outputs_ok, sf.get('status'))
    return dict(group=group, video=video, status=status, stage=sf.get('stage'),
                reason=sf.get('reason') or (r.stderr.strip().splitlines()[-1] if r.stderr.strip() else ''),
                error_type=sf.get('error_type'), returncode=r.returncode,
                output_path=str(out), outputs_ok=outputs_ok)


def engineering_gate(rows, expected=39):
    groups = {r['group'] for r in rows}
    sc = pd.Series([r['status'] for r in rows]).value_counts().to_dict()
    counts = dict(
        expected_groups=expected,
        discovered_groups=len(groups),
        input_valid_groups=len(groups),
        executed_groups=len(groups),
        completed_groups=sc.get('COMPLETED_VALID', 0),
        quality_limited_groups=sc.get('COMPLETED_LIMITED', 0),
        insufficient_evidence_groups=sc.get('INSUFFICIENT_EVIDENCE', 0),
        blocked_groups=sc.get('BLOCKED_QUALITY_GATE', 0),
        invalid_input_groups=sc.get('INVALID_INPUT', 0),
        implementation_error_groups=len({r['group'] for r in rows if r['status'] == 'IMPLEMENTATION_ERROR'}),
        missing_output_groups=len({r['group'] for r in rows
                                   if not r['outputs_ok']
                                   and r['status'] not in ('INSUFFICIENT_EVIDENCE', 'BLOCKED_QUALITY_GATE', 'INVALID_INPUT')}),
        unexpected_exception_count=sum(1 for r in rows if r['status'] == 'IMPLEMENTATION_ERROR'))
    ok = (counts['discovered_groups'] == expected and
          counts['input_valid_groups'] == expected and
          counts['executed_groups'] == expected and
          counts['implementation_error_groups'] == 0 and
          counts['missing_output_groups'] == 0 and
          counts['unexpected_exception_count'] == 0)
    return ok, counts


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--prepared-root', required=True)
    ap.add_argument('--output-root', required=True)
    ap.add_argument('--work-root', required=True)
    a = ap.parse_args()
    d = pd.read_csv(REG)
    groups = sorted(d.group.unique())
    rows = []
    for g in groups:
        for v in sorted(d[d.group == g].video.unique()):
            vid = Path(v).stem.split('_')[-1]
            rows.append(run_video(g, vid, a.prepared_root, a.output_root, a.work_root))
    ok, counts = engineering_gate(rows, expected=39)
    out = Path(a.output_root)
    pd.DataFrame(rows).to_csv(out/'EXECUTION_ROWS.csv', index=False)
    (out/'VALIDATION_39.json').write_text(json.dumps(dict(
        ENGINEERING_39_PASS=bool(ok), **counts,
        rows=len(rows), status_counts=pd.DataFrame(rows).status.value_counts().to_dict()), indent=1))
    pd.DataFrame(rows).to_csv(out/'EXECUTION_FAILURES.csv', index=False)
    print(json.dumps(dict(ENGINEERING_39_PASS=ok, **counts)))


if __name__ == '__main__':
    main()
```
